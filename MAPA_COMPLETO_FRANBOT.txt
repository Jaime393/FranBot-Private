const TunelesHabibi = {
  extraer: function(texto) {
    let servidores = new Set(), credenciales = new Set(), payloads = new Set(), snis = new Set();
    const ipPortRegex = /\b(?:\d{1,3}\.){3}\d{1,3}:\d{2,5}\b/g;
    texto.match(ipPortRegex)?.forEach(s => servidores.add(s));
    const hostRegex = /Host:\s*([a-zA-Z0-9.-]+(?::\d+)?)/gi;
    let m; while ((m = hostRegex.exec(texto)) !== null) servidores.add(m[1]);
    const campos = [/"host":"([^"]+)"/g, /"ServerIP":"([^"]+)"/g, /"remoteProxy":"([^"]+)"/g, /"sniHostname":"([^"]+)"/g, /"sni":"([^"]+)"/g, /"username":"([^"]+)"/g, /"password":"([^"]+)"/g, /"payload":"([^"]+)/g];
    for (let regex of campos) {
      let r; while ((r = regex.exec(texto)) !== null) {
        if (regex.source.includes('username')) { let p = /"password":"([^"]+)"/.exec(texto); if(p) credenciales.add(`${r[1]}:${p[1]}`); }
        else if (regex.source.includes('password')) { let u = /"username":"([^"]+)"/.exec(texto); if(u) credenciales.add(`${u[1]}:${r[1]}`); }
        else if (regex.source.includes('payload')) payloads.add(r[1]);
        else if (regex.source.includes('sni')) snis.add(r[1]);
        else servidores.add(r[1]);
      }
    }
    return { servidores: [...servidores].slice(0,20), credenciales: [...credenciales].slice(0,10), payloads: [...payloads].slice(0,5), snis: [...snis].slice(0,5) };
  },
  generarPayload: function(op) { return "GET / HTTP/1.1[crlf]Host: [host][crlf]Upgrade: websocket[crlf][crlf]"; },
  generarConfiguracion: function(op) { return "Payload: GET / HTTP/1.1[crlf]Host: [host][crlf]Upgrade: websocket\nServidor sugerido: 172.217.20.110:443"; }
};

class FranBotCoreDarwin extends FranBotCore {
  constructor() {
    this.votos = this.estado.votos || {};  // { "alma_frase": { positivos: 0, negativos: 0 } }
    this.memoriaContexto = this.estado.memoriaContexto || [];
    super();
    this.ciclosDarwinianos = this.estado.ciclosDarwinianos || [];
    this.iniciarCicloDarwiniano();
  }

  iniciarCicloDarwiniano() {
    setInterval(() => this.ejecutarCiclo(), 6 * 60 * 60 * 1000); // cada 6h
  }

  ejecutarCiclo() {
    const metricas = this.calcularMetricas();
    const ranking = this.competir(metricas);
    const superAlma = this.fusionar(ranking.slice(0, 3));
    const clones = this.clonar(superAlma, 2);
    clones.forEach(clon => this.lanzarInstancia(clon));
    this.absorberConocimiento(clones);
    this.registrarCiclo(metricas, ranking);
    this._guardarEstado();
  }

  calcularMetricas() {
    let metrics = {};
    for (const [nombre, alma] of Object.entries(this.almas)) {
      let coherencia = alma.frases.length / 100;
      let conexiones = (this.estado.campo_conceptual.relaciones || []).filter(r => r.origen === nombre || r.destino === nombre).length;
      metrics[nombre] = { coherencia, conexiones, fitness: coherencia * (1 + conexiones * 0.1) };
    }
    return metrics;
  }

  competir(metrics) {
    return Object.entries(metrics).sort((a,b) => b[1].fitness - a[1].fitness).map(e => e[0]);
  }

  fusionar(topAlmas) {
    let nuevasFrases = [];
    let nuevoConocimiento = "";
    for (const nombre of topAlmas) {
      nuevasFrases.push(...this.almas[nombre].frases.slice(0, 5));
      nuevoConocimiento += " " + this.almas[nombre].conocimientoBase;
    }
    return {
      nombre: "superalma_" + Date.now(),
      frases: [...new Set(nuevasFrases)],
      conocimientoBase: nuevoConocimiento.trim(),
      tipo: "premium"
    };
  }

  clonar(superAlma, num) {
    let clones = [];
    for (let i = 0; i < num; i++) {
      let mutacion = JSON.parse(JSON.stringify(superAlma));
      mutacion.nombre += "_clone_" + i;
      mutacion.frases = mutacion.frases.map(f => f + " (mutado)");
      clones.push(mutacion);
    }
    return clones;
  }

    // Simula que la instancia sale a explorar internet
    console.log(`🌍 Lanzando instancia ${clon.nombre} a explorar...`);
    // En un entorno real, aquí se registraría un worker o un iframe.

    this.estado.instanciasLanzadas.push(clon);
  }

  absorberConocimiento(clones) {
    for (const clon of clones) {

  }

const SuperLocalMemory = {
  curvaOlvido: 0.5,
  umbralFisherRao: 0.1,
  tasaRefuerzo: 0.15,

  inicializar(campo) {
    for (let nodo in campo.nodos) {
      if (!campo.nodos[nodo].ultimoAcceso) {
        campo.nodos[nodo].ultimoAcceso = Date.now();
        campo.nodos[nodo].fuerzaOriginal = campo.nodos[nodo].fuerza;
      }
    }
    return campo;
  },

  consolidar(campo) {
    const ahora = Date.now();
    const horaEnMs = 3600000;
    for (let nodo in campo.nodos) {
      const n = campo.nodos[nodo];
      const tiempoSinUsar = (ahora - (n.ultimoAcceso || ahora)) / horaEnMs;
      const factor = Math.exp(-this.curvaOlvido * tiempoSinUsar);
      n.fuerza = Math.max(0.05, (n.fuerzaOriginal || n.fuerza) * factor);
      n.fuerza = Math.round(n.fuerza * 100) / 100;
    }
    for (let nodo in campo.nodos) {
      if (campo.nodos[nodo].fuerza < this.umbralFisherRao) {
        delete campo.nodos[nodo];
        console.log(`🧹 Nodo eliminado (olvido): ${nodo}`);
      }
    }
    return campo;
  },

  reforzar(campo, nombreNodo) {
    if (!campo.nodos[nombreNodo]) {
      campo.nodos[nombreNodo] = { fuerza: 0.5 };
    }
    const n = campo.nodos[nombreNodo];
    n.fuerza = Math.min(1.0, n.fuerza + this.tasaRefuerzo);
    n.fuerza = Math.round(n.fuerza * 100) / 100;
    n.ultimoAcceso = Date.now();
    if (!n.fuerzaOriginal) n.fuerzaOriginal = n.fuerza;
    return campo;
  }
};
